function(get_renderer_unit_lib)
    list(TRANSFORM renderer_unit_SRC PREPEND "${CMAKE_SOURCE_DIR}/src/renderer/")
    add_library(renderer_unit_lib SHARED "${renderer_unit_SRC}")
    pkg_check_modules(RENDERER_UNIT_DEPS REQUIRED IMPORTED_TARGET cairomm-1.16 pangoft2 pangomm-2.48)
    target_link_libraries(renderer_unit_lib colors_unit_lib statics_unit_lib PkgConfig::RENDERER_UNIT_DEPS 2Geom::2geom GLibmm::GLibmm)
    make_target_unit_testable(renderer_unit_lib)
endfunction(get_renderer_unit_lib)
